{{- /* Returns the sessions from data/sessions.yml, sorted by start time and enriched with track colour, speaker pages, talk page and absolute times. Sessions without start/end are "announced but not scheduled": they still get speaker pages, talk pages and agenda listing, but no timetable slot. */ -}} {{- $data := site.Data.sessions -}} {{- $out := slice -}} {{- if and $data $data.enable $data.sessions -}} {{- $tz := $data.timezone | default "Europe/Rome" -}} {{- $tracks := dict -}} {{- range $data.tracks }}{{ $tracks = merge $tracks (dict .name .) }}{{ end -}} {{- $svcColor := $data.serviceColor | default "ink" -}} {{- $scheduled := slice -}} {{- $unscheduled := slice -}} {{- range $data.sessions -}} {{- $speakers := slice -}} {{- range .speakers }}{{ with site.GetPage (printf "/speakers/%s" .) }}{{ $speakers = $speakers | append . }}{{ end }}{{ end -}} {{- $url := "" -}} {{- if not .service }}{{ with site.GetPage (printf "/agenda/%s" .id) }}{{ $url = .RelPermalink }}{{ end }}{{ end -}} {{- $fallback := "ink" -}} {{- if .service }}{{ $fallback = $svcColor }}{{ end -}} {{- $track := index $tracks (.track | default "") | default dict -}} {{- $s := merge . (dict "color" ($track.color | default $fallback) "dark" ($track.dark | default false) "speakerPages" $speakers "url" $url ) -}} {{- if and .start .end -}} {{- $start := time.AsTime (printf "%sT%s:00" $data.date .start) $tz -}} {{- $end := time.AsTime (printf "%sT%s:00" $data.date .end) $tz -}} {{- $s = merge $s (dict "startTime" $start "endTime" $end "minutes" (int ($end.Sub $start).Minutes) ) -}} {{- $scheduled = $scheduled | append $s -}} {{- else -}} {{- $unscheduled = $unscheduled | append $s -}} {{- end -}} {{- end -}} {{- range sort $scheduled "start" }}{{ $out = $out | append . }}{{ end -}} {{- range $unscheduled }}{{ $out = $out | append . }}{{ end -}} {{- end -}} {{- return $out -}}
